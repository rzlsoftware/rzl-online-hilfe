        // Apps' 20 KB staticwebapp.config.json route limit (see

        var path = window.location.pathname;
        var candidates = [path];
        try {
          candidates.push(decodeURIComponent(path));
        } catch (error) {
          // ignore malformed percent-escapes
        }
        if (!path.endsWith('/')) candidates.push(path + '/');
        fetch('/legacy-redirects.json')
          .then(function (response) {
            return response.ok ? response.json() : null;
          })
          .then(function (map) {
            if (!map) return;
            for (var i = 0; i < candidates.length; i += 1) {
              var target = map[candidates[i]];
              if (target) {
                window.location.replace(target);
                return;
              }
            }
          })
          .catch(function () {
            // No redirect map available; leave the 404 message in place.
